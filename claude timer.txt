import React, { useState, useEffect, useRef } from 'react';
import { Play, Pause, RotateCcw, Plus, Trash2, Edit2, Check, X, ChevronUp, ChevronDown, Settings } from 'lucide-react';

export default function MultiTimer() {
  const [timers, setTimers] = useState([
    { id: 1, name: 'Study', minutes: 25, seconds: 0 },
    { id: 2, name: 'Break', minutes: 5, seconds: 0 },
  ]);
  const [currentTimerIndex, setCurrentTimerIndex] = useState(null);
  const [timeLeft, setTimeLeft] = useState(0);
  const [isPaused, setIsPaused] = useState(false);
  const [isRunning, setIsRunning] = useState(false);
  const [repeatAll, setRepeatAll] = useState(false);
  const [nextId, setNextId] = useState(3);
  const [editingId, setEditingId] = useState(null);
  const [editName, setEditName] = useState('');
  const [editMinutes, setEditMinutes] = useState(0);
  const [editSeconds, setEditSeconds] = useState(0);
  const [alarmVolume, setAlarmVolume] = useState(0.5);
  const [alarmDuration, setAlarmDuration] = useState(3);
  const [showSettings, setShowSettings] = useState(false);
  const audioContextRef = useRef(null);
  const alarmTimeoutRef = useRef(null);

  useEffect(() => {
    if (!isRunning || isPaused || currentTimerIndex === null) return;

    const interval = setInterval(() => {
      setTimeLeft(prev => {
        if (prev <= 1) {
          playAlert();
          moveToNextTimer();
          return 0;
        }
        return prev - 1;
      });
    }, 1000);

    return () => clearInterval(interval);
  }, [isRunning, isPaused, currentTimerIndex]);

  const playAlert = () => {
    if (alarmTimeoutRef.current) {
      clearTimeout(alarmTimeoutRef.current);
    }

    if (!audioContextRef.current) {
      audioContextRef.current = new (window.AudioContext || window.webkitAudioContext)();
    }

    const ctx = audioContextRef.current;
    const startTime = ctx.currentTime;
    const duration = alarmDuration;

    const beep = (time, frequency) => {
      const oscillator = ctx.createOscillator();
      const gainNode = ctx.createGain();

      oscillator.connect(gainNode);
      gainNode.connect(ctx.destination);

      oscillator.frequency.value = frequency;
      oscillator.type = 'sine';

      gainNode.gain.setValueAtTime(alarmVolume, time);
      gainNode.gain.exponentialRampToValueAtTime(0.01, time + 0.3);

      oscillator.start(time);
      oscillator.stop(time + 0.3);
    };

    const beepInterval = 0.5;
    const numBeeps = Math.floor(duration / beepInterval);

    for (let i = 0; i < numBeeps; i++) {
      beep(startTime + i * beepInterval, 800);
    }
  };

  const moveToNextTimer = () => {
    const nextIndex = currentTimerIndex + 1;
    
    if (nextIndex >= timers.length) {
      if (repeatAll) {
        setCurrentTimerIndex(0);
        setTimeLeft(timers[0].minutes * 60 + timers[0].seconds);
      } else {
        setIsRunning(false);
        setCurrentTimerIndex(null);
        setTimeLeft(0);
      }
    } else {
      setCurrentTimerIndex(nextIndex);
      setTimeLeft(timers[nextIndex].minutes * 60 + timers[nextIndex].seconds);
    }
  };

  const startSequence = () => {
    if (timers.length === 0) return;
    setCurrentTimerIndex(0);
    setTimeLeft(timers[0].minutes * 60 + timers[0].seconds);
    setIsRunning(true);
    setIsPaused(false);
  };

  const togglePause = () => {
    setIsPaused(!isPaused);
  };

  const resetSequence = () => {
    setIsRunning(false);
    setIsPaused(false);
    setCurrentTimerIndex(null);
    setTimeLeft(0);
  };

  const addTimer = () => {
    setTimers([...timers, {
      id: nextId,
      name: 'New Timer',
      minutes: 10,
      seconds: 0
    }]);
    setNextId(nextId + 1);
  };

  const deleteTimer = (id) => {
    const newTimers = timers.filter(timer => timer.id !== id);
    setTimers(newTimers);
    
    if (isRunning) {
      const deletedIndex = timers.findIndex(t => t.id === id);
      if (deletedIndex === currentTimerIndex) {
        resetSequence();
      } else if (deletedIndex < currentTimerIndex) {
        setCurrentTimerIndex(currentTimerIndex - 1);
      }
    }
  };

  const moveTimer = (index, direction) => {
    if (isRunning) return;
    
    const newTimers = [...timers];
    const newIndex = direction === 'up' ? index - 1 : index + 1;
    
    if (newIndex < 0 || newIndex >= timers.length) return;
    
    [newTimers[index], newTimers[newIndex]] = [newTimers[newIndex], newTimers[index]];
    setTimers(newTimers);
  };

  const startEdit = (timer) => {
    if (isRunning) return;
    setEditingId(timer.id);
    setEditName(timer.name);
    setEditMinutes(timer.minutes);
    setEditSeconds(timer.seconds);
  };

  const saveEdit = () => {
    setTimers(timers.map(timer =>
      timer.id === editingId
        ? { ...timer, name: editName, minutes: editMinutes, seconds: editSeconds }
        : timer
    ));
    setEditingId(null);
  };

  const cancelEdit = () => {
    setEditingId(null);
  };

  const formatTime = (totalSeconds) => {
    const mins = Math.floor(totalSeconds / 60);
    const secs = totalSeconds % 60;
    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  };

  const getTotalTime = () => {
    return timers.reduce((total, timer) => total + (timer.minutes * 60 + timer.seconds), 0);
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-blue-50 to-indigo-100 p-6">
      <div className="max-w-4xl mx-auto">
        <div className="flex justify-between items-center mb-2">
          <h1 className="text-4xl font-bold text-gray-800">Sequential Timer</h1>
          <button
            onClick={() => setShowSettings(!showSettings)}
            className="p-3 bg-white rounded-lg shadow-md hover:bg-gray-50 transition-colors"
            title="Alarm Settings"
          >
            <Settings size={24} className="text-gray-700" />
          </button>
        </div>
        <p className="text-center text-gray-600 mb-8">
          Total Duration: {formatTime(getTotalTime())}
        </p>

        {/* Alarm Settings Panel */}
        {showSettings && (
          <div className="bg-white rounded-lg shadow-md p-6 mb-6">
            <h2 className="text-xl font-semibold text-gray-800 mb-4">Alarm Settings</h2>
            <div className="space-y-4">
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  Volume: {Math.round(alarmVolume * 100)}%
                </label>
                <input
                  type="range"
                  min="0"
                  max="1"
                  step="0.01"
                  value={alarmVolume}
                  onChange={(e) => setAlarmVolume(parseFloat(e.target.value))}
                  className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer"
                />
              </div>
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  Duration: {alarmDuration} seconds
                </label>
                <input
                  type="range"
                  min="1"
                  max="10"
                  step="0.5"
                  value={alarmDuration}
                  onChange={(e) => setAlarmDuration(parseFloat(e.target.value))}
                  className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer"
                />
              </div>
              <button
                onClick={playAlert}
                className="px-4 py-2 bg-blue-500 hover:bg-blue-600 text-white rounded-md font-semibold transition-colors"
              >
                Test Alarm
              </button>
            </div>
          </div>
        )}

        {/* Current Timer Display */}
        {isRunning && currentTimerIndex !== null && (
          <div className="bg-white rounded-lg shadow-lg p-8 mb-6">
            <div className="text-center">
              <h2 className="text-3xl font-bold text-gray-800 mb-2">
                {timers[currentTimerIndex].name}
              </h2>
              <p className="text-sm text-gray-500 mb-4">
                Timer {currentTimerIndex + 1} of {timers.length}
              </p>
              <div className={`text-7xl font-bold mb-6 ${
                timeLeft <= 10 ? 'text-red-500' : 'text-blue-600'
              }`}>
                {formatTime(timeLeft)}
              </div>
              <div className="flex gap-3 justify-center">
                <button
                  onClick={togglePause}
                  className={`flex items-center gap-2 px-8 py-4 rounded-md font-semibold text-lg transition-colors ${
                    isPaused
                      ? 'bg-blue-500 hover:bg-blue-600 text-white'
                      : 'bg-yellow-500 hover:bg-yellow-600 text-white'
                  }`}
                >
                  {isPaused ? (
                    <>
                      <Play size={24} />
                      Resume
                    </>
                  ) : (
                    <>
                      <Pause size={24} />
                      Pause
                    </>
                  )}
                </button>
                <button
                  onClick={resetSequence}
                  className="flex items-center gap-2 px-8 py-4 bg-gray-500 hover:bg-gray-600 text-white rounded-md font-semibold text-lg transition-colors"
                >
                  <RotateCcw size={24} />
                  Stop
                </button>
              </div>
            </div>
          </div>
        )}

        {/* Control Panel */}
        {!isRunning && (
          <div className="bg-white rounded-lg shadow-md p-6 mb-6">
            <div className="flex items-center justify-between mb-4">
              <label className="flex items-center gap-3 cursor-pointer">
                <input
                  type="checkbox"
                  checked={repeatAll}
                  onChange={(e) => setRepeatAll(e.target.checked)}
                  className="w-5 h-5 text-blue-600 rounded focus:ring-2 focus:ring-blue-500"
                />
                <span className="text-lg font-semibold text-gray-700">Repeat Entire Sequence</span>
              </label>
              <button
                onClick={startSequence}
                disabled={timers.length === 0}
                className="flex items-center gap-2 px-8 py-4 bg-green-500 hover:bg-green-600 disabled:bg-gray-300 disabled:cursor-not-allowed text-white rounded-md font-semibold text-lg transition-colors"
              >
                <Play size={24} />
                Start Sequence
              </button>
            </div>
          </div>
        )}

        {/* Timer List */}
        <div className="space-y-3 mb-6">
          {timers.map((timer, index) => (
            <div
              key={timer.id}
              className={`bg-white rounded-lg shadow-md p-4 transition-all ${
                currentTimerIndex === index && isRunning ? 'ring-2 ring-blue-500' : ''
              }`}
            >
              {editingId === timer.id ? (
                <div className="space-y-3">
                  <input
                    type="text"
                    value={editName}
                    onChange={(e) => setEditName(e.target.value)}
                    className="w-full px-3 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
                    placeholder="Timer name"
                  />
                  <div className="flex gap-4 items-center">
                    <div className="flex items-center gap-2">
                      <label className="text-sm text-gray-600">Minutes:</label>
                      <input
                        type="number"
                        value={editMinutes}
                        onChange={(e) => setEditMinutes(Math.max(0, parseInt(e.target.value) || 0))}
                        className="w-20 px-3 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
                        min="0"
                      />
                    </div>
                    <div className="flex items-center gap-2">
                      <label className="text-sm text-gray-600">Seconds:</label>
                      <input
                        type="number"
                        value={editSeconds}
                        onChange={(e) => setEditSeconds(Math.max(0, Math.min(59, parseInt(e.target.value) || 0)))}
                        className="w-20 px-3 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
                        min="0"
                        max="59"
                      />
                    </div>
                  </div>
                  <div className="flex gap-2">
                    <button
                      onClick={saveEdit}
                      className="flex items-center gap-2 px-4 py-2 bg-green-500 text-white rounded-md hover:bg-green-600 transition-colors"
                    >
                      <Check size={18} />
                      Save
                    </button>
                    <button
                      onClick={cancelEdit}
                      className="flex items-center gap-2 px-4 py-2 bg-gray-500 text-white rounded-md hover:bg-gray-600 transition-colors"
                    >
                      <X size={18} />
                      Cancel
                    </button>
                  </div>
                </div>
              ) : (
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-4">
                    <div className="flex flex-col gap-1">
                      <button
                        onClick={() => moveTimer(index, 'up')}
                        disabled={index === 0 || isRunning}
                        className="p-1 text-gray-600 hover:bg-gray-100 rounded disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
                      >
                        <ChevronUp size={16} />
                      </button>
                      <button
                        onClick={() => moveTimer(index, 'down')}
                        disabled={index === timers.length - 1 || isRunning}
                        className="p-1 text-gray-600 hover:bg-gray-100 rounded disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
                      >
                        <ChevronDown size={16} />
                      </button>
                    </div>
                    <div>
                      <div className="flex items-center gap-2">
                        <span className="text-lg font-semibold text-gray-800">{timer.name}</span>
                        <span className="text-2xl font-mono text-blue-600">
                          {formatTime(timer.minutes * 60 + timer.seconds)}
                        </span>
                      </div>
                      {currentTimerIndex === index && isRunning && (
                        <span className="text-sm text-blue-600 font-semibold">Currently Running</span>
                      )}
                    </div>
                  </div>
                  <div className="flex gap-2">
                    <button
                      onClick={() => startEdit(timer)}
                      disabled={isRunning}
                      className="p-2 text-gray-600 hover:bg-gray-100 rounded-md transition-colors disabled:opacity-30 disabled:cursor-not-allowed"
                    >
                      <Edit2 size={18} />
                    </button>
                    <button
                      onClick={() => deleteTimer(timer.id)}
                      disabled={isRunning}
                      className="p-2 text-red-600 hover:bg-red-50 rounded-md transition-colors disabled:opacity-30 disabled:cursor-not-allowed"
                    >
                      <Trash2 size={18} />
                    </button>
                  </div>
                </div>
              )}
            </div>
          ))}
        </div>

        <button
          onClick={addTimer}
          disabled={isRunning}
          className="w-full flex items-center justify-center gap-2 px-6 py-4 bg-indigo-500 hover:bg-indigo-600 disabled:bg-gray-300 disabled:cursor-not-allowed text-white rounded-lg font-semibold text-lg transition-colors shadow-md"
        >
          <Plus size={24} />
          Add New Timer
        </button>
      </div>
    </div>
  );
}
